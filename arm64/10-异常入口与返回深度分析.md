## 1. 概述

ARM64 异常模型是 QEMU 中最复杂的模拟逻辑之一，涵盖：

- **异常入口 (Exception Entry)**：保存当前状态 → 确定向量偏移 → 设置新 PSTATE → 跳转到向量表
- **异常返回 (Exception Return)**：ERET 指令 → 恢复 SPSR 到 PSTATE → 恢复 ELR 到 PC → 切换 EL

QEMU 实现的关键设计：
1. **翻译时 (TCG 前端)**：SVC/HVC/SMC/ERET 等指令在 `translate-a64.c` 中翻译为 helper 调用
2. **运行时 (Helper)**：`raise_exception()` 设置异常参数后通过 `cpu_loop_exit()` 退出 TB
3. **异常分发**：`arm_cpu_do_interrupt()` 是统一入口，根据目标 EL 是否 AArch64 分派到不同处理函数

```
指令执行 (TB)
  │
  ├── SVC/HVC/SMC → raise_exception() → cpu_loop_exit()
  ├── 内存故障     → tlb_helper.c → raise_exception()
  ├── 系统寄存器陷入 → handle_sys() → raise_exception()
  │
  ▼
cpu_exec() 捕获异常
  │
  ▼
arm_cpu_do_interrupt()         [helper.c:9469]
  ├── PSCI call → arm_handle_psci_call()
  ├── Semihosting → tcg_handle_semihosting()
  ├── AArch64 → arm_cpu_do_interrupt_aarch64()   [helper.c:9198]
  └── AArch32 → arm_cpu_do_interrupt_aarch32()   [helper.c:8875]
```

## 2. 异常触发与目标 EL 确定

### 2.1 同步异常目标 EL：exception_target_el()

```c
// op_helper.c:32-44
int exception_target_el(CPUARMState *env)
{
    int target_el = MAX(1, arm_current_el(env));
    // Secure EL1 在 EL3 为 AArch32 时重定向到 EL3
    if (arm_is_secure(env) && !arm_el_is_aa64(env, 3) && target_el == 1) {
        target_el = 3;
    }
    return target_el;
}
```

**规则**：同步异常至少进入 EL1（永远不会停留在 EL0）。

### 2.2 raise_exception() — 异常触发核心

```c
// op_helper.c:47-69
void raise_exception(CPUARMState *env, uint32_t excp,
                     uint64_t syndrome, uint32_t target_el)
{
    // HCR_EL2.TGE 重定向：EL1 → EL2
    if (target_el == 1 && (arm_hcr_el2_eff(env) & HCR_TGE)) {
        target_el = 2;
        // FP/SIMD 陷入变为 uncategorized
        if (syn_get_ec(syndrome) == EC_ADVSIMDFPACCESSTRAP) {
            syndrome = syn_uncategorized();
        }
    }
    cs->exception_index = excp;
    env->exception.syndrome = syndrome;
    env->exception.target_el = target_el;
    cpu_loop_exit(cs);  // 不返回！跳回 cpu_exec() 主循环
}
```

**关键行为**：
- `HCR_TGE=1` 将所有 NS EL1 异常重定向到 EL2
- FP/SIMD 陷入在重定向时综合征变为 uncategorized (DDI0478C.a D1.10.4)
- `cpu_loop_exit()` 通过 `longjmp` 回到 `cpu_exec()` 主循环

### 2.3 物理中断目标 EL：arm_phys_excp_target_el()

```c
// helper.c:8369-8420
uint32_t arm_phys_excp_target_el(CPUState *cs, uint32_t excp_idx,
                                  uint32_t cur_el, bool secure)
```

通过 6 维查找表 `target_el_table[is64][scr][rw][hcr][secure][cur_el]` 确定目标 EL：

| 因素 | 来源 | 说明 |
|------|------|------|
| `is64` | `ARM_FEATURE_AARCH64` | 最高 EL 是否 AArch64 |
| `scr` | `SCR_EL3.IRQ/FIQ/EA` | EL3 是否路由该中断 |
| `rw` | `SCR_EL3.RW` | EL2/EL1 寄存器宽度 |
| `hcr` | `HCR_EL2.IMO/FMO/AMO` + `TGE` | EL2 是否路由该中断 |
| `secure` | 当前安全状态 | Secure/Non-secure |
| `cur_el` | 当前 EL | 0-3 |

```c
// helper.c:8393-8413
switch (excp_idx) {
case EXCP_IRQ:
case EXCP_NMI:
    scr = (env->cp15.scr_el3 & SCR_IRQ);  // SCR.IRQ → EL3
    hcr = hcr_el2 & HCR_IMO;              // HCR.IMO → EL2
    break;
case EXCP_FIQ:
    scr = (env->cp15.scr_el3 & SCR_FIQ);  // SCR.FIQ → EL3
    hcr = hcr_el2 & HCR_FMO;              // HCR.FMO → EL2
    break;
default:  // SError
    scr = (env->cp15.scr_el3 & SCR_EA);   // SCR.EA → EL3
    hcr = hcr_el2 & HCR_AMO;              // HCR.AMO → EL2
    break;
}
// TGE 强制中断到 EL2
hcr |= (hcr_el2 & HCR_TGE) != 0;

target_el = target_el_table[is64][scr][rw][hcr][secure][cur_el];
```

## 3. 同步异常触发路径

### 3.1 SVC/HVC/SMC（翻译阶段）

```c
// translate-a64.c:3155-3204

// SVC — 目标 EL 由 exception_target_el() 决定
static bool trans_SVC(DisasContext *s, arg_i *a)
{
    uint32_t syndrome = syn_aa64_svc(a->imm);
    if (s->fgt_svc) {  // Fine-grained trap
        gen_exception_insn_el(s, 0, EXCP_UDEF, syndrome, 2);
        return true;
    }
    gen_ss_advance(s);   // 单步状态机前进（架构要求）
    gen_exception_insn(s, 4, EXCP_SWI, syndrome);
    return true;
}

// HVC — 目标 EL2（EL3 时留在 EL3）
static bool trans_HVC(DisasContext *s, arg_i *a)
{
    int target_el = s->current_el == 3 ? 3 : 2;
    gen_a64_update_pc(s, 0);
    gen_helper_pre_hvc(tcg_env);  // 运行时检查是否陷入为 UDEF
    gen_ss_advance(s);
    gen_exception_insn_el(s, 4, EXCP_HVC, syn_aa64_hvc(a->imm), target_el);
    return true;
}

// SMC — 固定目标 EL3
static bool trans_SMC(DisasContext *s, arg_i *a)
{
    gen_a64_update_pc(s, 0);
    gen_helper_pre_smc(tcg_env, tcg_constant_i32(syn_aa64_smc(a->imm)));
    gen_ss_advance(s);
    gen_exception_insn_el(s, 4, EXCP_SMC, syn_aa64_smc(a->imm), 3);
    return true;
}
```

**架构要求**：SVC/HVC/SMC 必须在触发异常前推进单步状态机 (`gen_ss_advance`)。

## 4. 异步中断分发

### 4.1 arm_cpu_exec_interrupt() — 中断优先级与分发

```c
// cpu-irq.c:171-273
bool arm_cpu_exec_interrupt(CPUState *cs, int interrupt_request)
```

中断按以下优先级检查（实现定义）：

```
┌─────────────────────────────────────────────────────┐
│ NMI 功能启用时 (FEAT_NMI + SCTLR.NMI)              │
│  1. EXCP_NMI    (物理 NMI)       → target_el=路由表│
│  2. EXCP_VINMI  (虚拟 NMI-IRQ)   → target_el=1    │
│  3. EXCP_VFNMI  (虚拟 NMI-FIQ)   → target_el=1    │
├─────────────────────────────────────────────────────┤
│ NMI 功能禁用时：NMI→IRQ, VINMI→VIRQ, VFNMI→VFIQ   │
├─────────────────────────────────────────────────────┤
│  4. EXCP_FIQ    (物理 FIQ)       → target_el=路由表│
│  5. EXCP_IRQ    (物理 IRQ)       → target_el=路由表│
│  6. EXCP_VIRQ   (虚拟 IRQ)       → target_el=1    │
│  7. EXCP_VFIQ   (虚拟 FIQ)       → target_el=1    │
│  8. EXCP_VSERR  (虚拟 SError)    → target_el=1    │
└─────────────────────────────────────────────────────┘
```

**关键细节**：
- 虚拟中断 (VIRQ/VFIQ/VSERR/VINMI/VFNMI) 固定目标 EL1
- 物理中断通过 `arm_phys_excp_target_el()` 查路由表
- `arm_excp_unmasked()` 检查 PSTATE.DAIF 掩码
- VSERR 接受时清除 `HCR_EL2.VSE` (cpu-irq.c:262)

```c
// cpu-irq.c:269-273 — 找到可送达的中断后
found:
    cs->exception_index = excp_idx;
    env->exception.target_el = target_el;
    cs->cc->tcg_ops->do_interrupt(cs);  // → arm_cpu_do_interrupt()
    return true;
```

## 5. 调试异常

```c
// debug.c:18-40
static int arm_debug_target_el(CPUARMState *env)
```

调试异常目标 EL 由 MDCR_EL2/MDCR_EL3 控制：
- MDCR_EL3.SDD → EL3 调试异常禁用
- MDCR_EL2.TDE → 调试异常路由到 EL2
- 否则默认 `MAX(1, cur_el)`

```c
// debug.c:42-60
void raise_exception_debug(CPUARMState *env, uint32_t excp, uint64_t syndrome)
{
    int debug_el = arm_debug_target_el(env);
    int cur_el = arm_current_el(env);
    // HCR.TGE 重定向
    if (debug_el == 1 && arm_hcr_el2_eff(env) & HCR_TGE) {
        debug_el = 2;
    }
    raise_exception(env, excp, syndrome, debug_el);
}
```

调试异常类型：
- 断点 (Breakpoint)：`syn_breakpoint()` — debug.c:437-538
- 监视点 (Watchpoint)：debug.c:546-650
- 单步 (Software Step)：`syn_swstep()` — debug.c:541-544
- BRK 指令：`syn_aa64_bkpt()` — translate-a64.c:3207-3210

## 6. 异常入口总控：arm_cpu_do_interrupt()

```c
// helper.c:9469-9530
void arm_cpu_do_interrupt(CPUState *cs)
{
    ARMCPU *cpu = ARM_CPU(cs);
    CPUARMState *env = &cpu->env;
    unsigned int new_el = env->exception.target_el;

    assert(!arm_feature(env, ARM_FEATURE_M));  // M-profile 有独立处理

    arm_log_exception(cs);

    // 1. PSCI 调用拦截（TCG 模式）
    if (tcg_enabled() && arm_is_psci_call(cpu, cs->exception_index)) {
        arm_handle_psci_call(cpu);
        return;
    }

    // 2. Semihosting 拦截
    if (cs->exception_index == EXCP_SEMIHOST) {
        tcg_handle_semihosting(cs);
        return;
    }

    // 3. 必须持有 BQL
    g_assert(bql_locked());

    // 4. EL 变更前 hook（通知 KVM 等）
    arm_call_pre_el_change_hook(cpu);

    // 5. 根据目标 EL 的执行状态分派
    assert(!excp_is_internal(cs->exception_index));
    if (arm_el_is_aa64(env, new_el)) {
        arm_cpu_do_interrupt_aarch64(cs);   // 9198 行
    } else {
        arm_cpu_do_interrupt_aarch32(cs);   // 8875 行
    }

    // 6. EL 变更后 hook
    arm_call_el_change_hook(cpu);

    // 7. 强制退出当前 TB（非 KVM）
    if (!kvm_enabled()) {
        cpu_set_interrupt(cs, CPU_INTERRUPT_EXITTB);
    }
}
```

**设计要点**：
- PSCI 和 Semihosting 是"伪异常"，拦截后直接处理不走正常异常路径
- `arm_el_is_aa64(env, new_el)` 检查目标 EL 的配置，而非当前执行状态
- `CPU_INTERRUPT_EXITTB` 确保异常后不继续执行旧 TB

## 8. 向量表偏移计算

VBAR_ELn 基地址加上异常类型和来源的偏移：

```c
// helper.c:9203
```

### 8.1 来源 EL 偏移

```
┌──────────────────┬────────────────────────────────────┐
│ 条件             │ 偏移                                │
├──────────────────┼────────────────────────────────────┤
│ 同级 EL, SP_EL0  │ +0x000                             │
│ 同级 EL, SP_ELx  │ +0x200  [PSTATE.SP=1]              │
│ 低级 EL, AArch64 │ +0x400                             │
│ 低级 EL, AArch32 │ +0x600                             │
└──────────────────┴────────────────────────────────────┘
```

实现代码：

```c
// helper.c:9217-9254
if (cur_el < new_el) {
    // 低级 EL 进入高级 EL
    bool is_aa64;
    switch (new_el) {
    case 3:
        is_aa64 = arm_scr_rw_eff(env);           // SCR.RW 决定
        break;
    case 2:
        hcr = arm_hcr_el2_eff(env);
        if ((hcr & (HCR_E2H | HCR_TGE)) != (HCR_E2H | HCR_TGE)) {
            is_aa64 = (hcr & HCR_RW) != 0;       // HCR.RW 决定
            break;
        }
        /* VHE+TGE: fall through to use is_a64() */
    case 1:
        is_aa64 = is_a64(env);                    // 当前执行状态
        break;
    }
    addr += is_aa64 ? 0x400 : 0x600;
} else {
    // 同级 EL
    if (pstate_read(env) & PSTATE_SP) {
        addr += 0x200;                            // 使用 SP_ELx
    }
    // GCS EXLOCK 仅在同级异常时设置
    if (is_a64(env) && (env->cp15.gcscr_el[new_el] & GCSCR_EXLOCKEN)) {
        new_mode |= PSTATE_EXLOCK;
    }
}
```

## 9. ESR 综合征生成与写入

### 9.1 综合征构建函数 (syndrome.h)

综合征是 32 位值，高 6 位为 Exception Class (EC)，低 25 位为 ISS：

```
  31  26 25 24                    0
 ┌──────┬──┬────────────────────────┐
 │  EC  │IL│         ISS            │
 └──────┴──┴────────────────────────┘
```

关键 EC 值定义于 `syndrome.h:31-77`，主要构建函数：

| 函数 | 位置 | 用途 |
|------|------|------|
| `syn_aa64_svc()` | syndrome.h:138-142 | SVC #imm16 |
| `syn_aa64_hvc()` | syndrome.h:144-148 | HVC #imm16 |
| `syn_aa64_smc()` | syndrome.h:150-154 | SMC #imm16 |
| `syn_insn_abort()` | syndrome.h:516-526 | 指令中止 |
| `syn_data_abort_no_iss()` | syndrome.h:546-556 | 数据中止（无 ISS） |
| `syn_data_abort_with_iss()` | syndrome.h:558-586 | 数据中止（有 ISS） |
| `syn_swstep()` | syndrome.h:611-620 | 软件单步 |
| `syn_breakpoint()` | syndrome.h:655-662 | 断点 |

### 11.4 pstate_read() / pstate_write() 详解

```c
// cpu.h:1607-1626
static inline uint64_t pstate_read(CPUARMState *env)
{
    // 组合分散存储的各 PSTATE 字段
    int ZF = (env->ZF == 0);
    return (env->NF & 0x80000000)              // N
         | (ZF << 30)                          // Z
         | (env->CF << 29)                     // C
         | ((env->VF & 0x80000000) >> 3)       // V
         | env->pstate                         // 其余位 (SP, EL, nRW, ...)
         | env->daif                           // DAIF 掩码
         | (env->btype << 10);                 // BTYPE
}

static inline void pstate_write(CPUARMState *env, uint64_t val)
{
    env->ZF = (~val) & PSTATE_Z;
    env->NF = val;
    env->CF = (val >> 29) & 1;
    env->VF = (val << 3) & 0x80000000;
    env->daif = val & PSTATE_DAIF;
    env->btype = (val >> 10) & 3;
    env->pstate = val & ~CACHED_PSTATE_BITS;   // 排除已缓存的位
}
```

**设计要点**：NZCV、DAIF、BTYPE 被分散缓存在独立字段中，以避免频繁的位操作。`pstate_read()/pstate_write()` 负责打包/解包。

### 11.5 AArch32 CPSR → SPSR 转换

```c
// helper.c:9139-9152

{
```

### 13.1 aarch64_save_sp() — 入口时保存

```c
{
```

## 16. FEAT_NV/NV2 对异常入口的影响

当 `HCR_EL2.NV=1` 时，EL1 的异常行为会发生变化：

## 17. FEAT_DoubleFault 重定向

## 21. 非法异常返回处理

当 ERET 检测到非法条件时跳转到 `illegal_return` 标签：

```c
// helper-a64.c:766-781
illegal_return:
    // 架构规定的行为：
    // 1. 设置 PSTATE.IL（非法执行状态位）
    env->pstate |= PSTATE_IL;

    // 2. 从 ELR 恢复 PC
    env->pc = new_pc;

    // 3. 仅恢复 NZCV + DAIF + ALLINT，保留其余 PSTATE
    spsr &= PSTATE_NZCV | PSTATE_DAIF | PSTATE_ALLINT;
    spsr |= pstate_read(env) & ~(PSTATE_NZCV | PSTATE_DAIF | PSTATE_ALLINT);
    pstate_write(env, spsr);

    // 4. 清除 SS（如非单步模式）
    if (!arm_singlestep_active(env)) {
        env->pstate &= ~PSTATE_SS;
    }
```

**非法返回不会**：
- 改变异常级别
- 改变执行状态（AArch32/AArch64）
- 改变栈指针选择

**非法返回条件汇总**：

| 条件 | 代码位置 | 说明 |
|------|---------|------|
| `el_from_spsr()` 返回 -1 | helper-a64.c:646-648 | SPSR.M 字段无效 |
| `new_el > cur_el` | helper-a64.c:650 | 试图返回到更高 EL |
| EL2 未实现 | helper-a64.c:650 | `new_el==2` 但 EL2 未启用 |
| RME 安全状态无效 | helper-a64.c:663-665 | EL3→ELx 安全状态禁止 |
| 寄存器宽度不匹配 | helper-a64.c:668-670 | 目标 EL 配置与 SPSR.nRW 矛盾 |

## 22. ELR → PC 恢复与 TBI

### 22.1 AArch64 TBI 处理

```c
// helper-a64.c:730-747
tbii = EX_TBFLAG_A64(env->hflags, TBII);
if ((tbii >> extract64(new_pc, 55, 1)) & 1) {
    // TBI 启用：根据地址空间范围截断/符号扩展
    int core_mmu_idx = arm_env_mmu_index(env);
    if (regime_has_2_ranges(core_to_aa64_mmu_idx(core_mmu_idx))) {
        new_pc = sextract64(new_pc, 0, 56);   // 双范围：符号扩展
    } else {
        new_pc = extract64(new_pc, 0, 56);     // 单范围：零扩展
    }
}
env->pc = new_pc;
```

**设计说明**：TBI 处理在 ERET helper 中延迟执行，因为需要先确定目标 EL 的 hflags 才能读取正确的 TBII 字段。

### 22.2 AArch32 PC 对齐

```c
// helper-a64.c:709-713
if (spsr & CPSR_T) {
    env->regs[15] = new_pc & ~0x1;   // Thumb: 半字对齐
} else {
    env->regs[15] = new_pc & ~0x3;   // ARM: 字对齐
}
```

## 23. AArch32 ↔ AArch64 状态切换

### 23.1 AArch32 → AArch64（异常入口）

```c
// helper.c:8476-8574 — aarch64_sync_32_to_64()
```

将 AArch32 的 banked 寄存器（R0-R14 在不同模式下的 bank）映射到 AArch64 的 X0-X30：
- R0-R7 → X0-X7（直接映射）
- R8-R14 → 根据 AArch32 模式的 bank 规则映射
- CPSR → 已通过 `cpsr_read_for_spsr_elx()` 转换
- condexec_bits 清零

### 23.2 AArch64 → AArch32（ERET 返回）

```c
// helper.c:8581-8605+ — aarch64_sync_64_to_32()
```

反向映射：X0-X14 → R0-R14（按目标 AArch32 模式选择正确的 bank）。

## 24. SVC/HVC/SMC 入口与返回

### 24.1 完整生命周期

```
┌─────────────────────────────────────────────────────────┐
│ SVC 生命周期                                            │
│                                                         │
│  EL0: SVC #imm16                                        │
│    ├── gen_ss_advance(s)        // 单步推进              │
│    ├── syn_aa64_svc(imm)        // 生成综合征            │
│    └── gen_exception_insn()     // → raise_exception()  │
│                                                         │
│  → cpu_exec() 捕获                                     │
│  → arm_cpu_do_interrupt()                               │
│  → arm_cpu_do_interrupt_aarch64()                       │
│    ├── VBAR_EL1 + 0x400         // 低级 EL, AArch64     │
│    ├── SPSR_EL1 = PSTATE        // 保存                 │
│    ├── ELR_EL1 = PC             // 保存返回地址          │
│    ├── ESR_EL1 = syn_aa64_svc() // 写入综合征           │
│    ├── PSTATE = DAIF|EL1h       // 新状态               │
│    └── PC = VBAR_EL1 + 0x400   // 跳转                 │
│                                                         │
│  EL1: 内核处理 SVC                                      │
│    └── ERET                                             │
│                                                         │
│  → HELPER(exception_return)                             │
│    ├── PSTATE ← SPSR_EL1       // 恢复                 │
│    ├── PC ← ELR_EL1            // 恢复                 │
│    └── 回到 EL0                                         │
└─────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────┐
│ HVC 生命周期                                            │
│                                                         │
│  EL1: HVC #imm16                                        │
│    ├── gen_helper_pre_hvc()     // 运行时陷入检查        │
│    ├── gen_ss_advance(s)                                │
│    └── target_el = 2 (或 EL3 时留 3)                    │
│                                                         │
│  → VBAR_EL2 + 0x400 (从低级 EL)                        │
│  → ESR_EL2 = syn_aa64_hvc(imm)                         │
│  → ERET 返回 EL1                                       │
└─────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────┐
│ SMC 生命周期                                            │
│                                                         │
│  EL1: SMC #imm16                                        │
│    ├── gen_helper_pre_smc()     // 检查 SCR.SMD 等      │
│    ├── gen_ss_advance(s)                                │
│    └── target_el = 3 (固定)                             │
│                                                         │
│  → VBAR_EL3 + 0x400                                    │
│  → ESR_EL3 = syn_aa64_smc(imm)                         │
│  → ERET 返回 EL1                                       │
└─────────────────────────────────────────────────────────┘
```

### 24.2 PSCI 拦截

在 `arm_cpu_do_interrupt()` 中，SMC/HVC 可能被识别为 PSCI 调用：

```c
// helper.c:9488-9493
if (tcg_enabled() && arm_is_psci_call(cpu, cs->exception_index)) {
    arm_handle_psci_call(cpu);
    return;  // 不走正常异常路径
}
```

PSCI 调用（如 CPU_ON、CPU_OFF、SYSTEM_RESET）由 QEMU 直接处理，不进入 guest 的异常向量。
